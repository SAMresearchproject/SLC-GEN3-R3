def run_sam_v4_1_engine(L, channel, carrier=None, guards=None):
    C_w = [0.5, 0.5, 0.5, 0.5, 0.5, 0.5]
    D_route = sum(C_w)
    L_w = 2 * ALPHA_H
    R = L_w * D_route

    A0 = 1 / (pi * R)
    A_share = 1 / R
    A_side = 1 / (2 * R)

    assert close(D_route, D)
    assert close(R, 2 * ALPHA_H * D)
    assert close(R * A_share, 1)
    assert close(2 * R * A_side, 1)
    assert not close(R * A0, 1)

    A_L = construct_accumulation_field(L)

    exposure = declare_exposure(channel, A_L)
    validate_exposure_declaration(exposure)

    exposure_type = type_exposure(exposure)

    if exposure_type == "seconds":
        readout = time_action_branch(exposure, carrier)

    elif exposure_type == "meters":
        readout = road_length_branch(exposure, carrier)

    elif exposure_type == "gradient":
        readout = force_potential_branch(exposure, carrier)

    elif exposure_type == "side_route":

            exposure,
            guards,
            R=R,
            A_share=A_share,
            A_side=A_side,
            D_route=D_route,

        readout = closed_topology_branch(exposure, guards)

    elif exposure_type == "inventory":
        readout = inventory_branch(exposure, guards)

    else:
        return reject("undeclared exposure type")

    ledger_status = LedgerWrite(
        readout,
        Gamma_res=guards.Gamma_res,
        X_c=guards.X_c,
