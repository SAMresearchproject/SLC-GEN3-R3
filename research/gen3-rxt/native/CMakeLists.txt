cmake_minimum_required(VERSION 3.24)
project(GEN3_RXT_R2 VERSION 0.1.0 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
find_package(CUDAToolkit 12.8 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(OpenSSL REQUIRED)
set(RXT_DEPS "${CMAKE_SOURCE_DIR}/deps/usr" CACHE PATH "Private native dependency prefix")
find_path(GMP_INCLUDE_DIR gmpxx.h HINTS "${RXT_DEPS}/include" REQUIRED)
find_path(GMP_C_INCLUDE_DIR gmp.h HINTS "${RXT_DEPS}/include/x86_64-linux-gnu" REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp HINTS "${RXT_DEPS}/include" REQUIRED)
find_library(GMP_LIBRARY gmp HINTS "${RXT_DEPS}/lib/x86_64-linux-gnu" REQUIRED)
find_library(GMPXX_LIBRARY gmpxx HINTS "${RXT_DEPS}/lib/x86_64-linux-gnu" REQUIRED)
add_executable(gen-rxt src/main.cpp src/engine.cpp src/core.cpp src/training.cpp src/transfer.cpp src/horizon.cpp src/campaign.cpp src/rh.cpp src/cuda.cu src/atom3d.cpp src/atom3d_cuda.cu src/service.cpp src/construction.cpp src/starbreaker.cpp src/knowledge.cpp src/joint.cpp src/starbreaker_cuda.cu)
target_include_directories(gen-rxt PRIVATE src ${GMP_INCLUDE_DIR} ${GMP_C_INCLUDE_DIR} ${JSON_INCLUDE_DIR})
target_link_libraries(gen-rxt PRIVATE CUDA::cudart OpenMP::OpenMP_CXX OpenSSL::Crypto ${GMPXX_LIBRARY} ${GMP_LIBRARY})
target_compile_options(gen-rxt PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3;-Wall;-Wextra;-Wpedantic> $<$<COMPILE_LANGUAGE:CUDA>:-O3;--ptxas-options=-v>)
set_target_properties(gen-rxt PROPERTIES BUILD_RPATH "${RXT_DEPS}/lib/x86_64-linux-gnu")
